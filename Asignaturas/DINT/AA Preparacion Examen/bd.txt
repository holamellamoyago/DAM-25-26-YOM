-- 1. Preparación del Entorno
DROP DATABASE IF EXISTS F1_History;
CREATE DATABASE IF NOT EXISTS F1_History;
USE F1_History;

-- 2. Equipos (Constructores)
CREATE TABLE IF NOT EXISTS constructors (
    constructor_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL UNIQUE,
    nationality VARCHAR(50),
    active BOOLEAN DEFAULT TRUE
) ENGINE=InnoDB;

-- 3. Pilotos
CREATE TABLE IF NOT EXISTS drivers (
    driver_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    driver_code CHAR(3) UNIQUE, -- Ej: VER, HAM
    nationality VARCHAR(50),
    permanent_number INT,
    INDEX (last_name)
) ENGINE=InnoDB;

-- 4. Circuitos
CREATE TABLE IF NOT EXISTS circuits (
    circuit_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(100),
    country VARCHAR(100),
    track_length_km DECIMAL(5,3)
) ENGINE=InnoDB;

-- 5. Carreras
CREATE TABLE IF NOT EXISTS races (
    race_id INT PRIMARY KEY AUTO_INCREMENT,
    year INT NOT NULL,
    round INT NOT NULL,
    circuit_id INT,
    name VARCHAR(100),
    race_date DATE,
    FOREIGN KEY (circuit_id) REFERENCES circuits(circuit_id) ON DELETE CASCADE,
    UNIQUE KEY (year, round) -- Evita duplicar carreras en el mismo año/ronda
) ENGINE=InnoDB;

-- 6. Resultados (Historial, Victorias, Puntos)
CREATE TABLE IF NOT EXISTS results (
    result_id INT PRIMARY KEY AUTO_INCREMENT,
    race_id INT NOT NULL,
    driver_id INT NOT NULL,
    constructor_id INT NOT NULL,
    grid_position INT,
    final_position INT,
    points DECIMAL(4,1) DEFAULT 0.0,
    laps_completed INT,
    fastest_lap BOOLEAN DEFAULT FALSE,
    status VARCHAR(50) DEFAULT 'Finished',
    FOREIGN KEY (race_id) REFERENCES races(race_id),
    FOREIGN KEY (driver_id) REFERENCES drivers(driver_id),
    FOREIGN KEY (constructor_id) REFERENCES constructors(constructor_id),
    INDEX (final_position) -- Optimiza búsquedas de victorias
) ENGINE=InnoDB;
